# [Phase 1] Public support request false-positive tuning
# ----------------------------------------------------------------
# Scope:
# - POST /api/public/support-requests
# - JSON fields only
# Intent:
# - Allow normal and contextual security-study submissions to reach app logic

SecRule REQUEST_HEADERS:Host "@rx ^(?:www\.)?kj\.ac\.kr(?::\d+)?$" \
    "id:1000210,phase:1,pass,nolog,chain"
    SecRule REQUEST_METHOD "@streq POST" "chain"
    SecRule REQUEST_FILENAME "@streq /api/public/support-requests" "chain"
    SecRule REQUEST_HEADERS:Content-Type "@contains application/json" \
    "t:none,\
    ctl:ruleRemoveTargetByTag=attack-sqli;ARGS:category,\
    ctl:ruleRemoveTargetByTag=attack-sqli;ARGS:subject,\
    ctl:ruleRemoveTargetByTag=attack-sqli;ARGS:message,\
    ctl:ruleRemoveTargetByTag=attack-xss;ARGS:subject,\
    ctl:ruleRemoveTargetByTag=attack-xss;ARGS:message,\
    ctl:ruleRemoveTargetByTag=attack-rce;ARGS:subject,\
    ctl:ruleRemoveTargetByTag=attack-rce;ARGS:message,\
    ctl:ruleRemoveTargetByTag=attack-lfi;ARGS:subject,\
    ctl:ruleRemoveTargetByTag=attack-lfi;ARGS:message,\
    ctl:ruleRemoveTargetByTag=attack-sqli;ARGS:referenceUrl,\
    ctl:ruleRemoveTargetByTag=attack-xss;ARGS:referenceUrl,\
    msg:'PROJECT_SUPPORT_REQUEST_FP_TUNING'"
